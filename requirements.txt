# ---- ядро ----
pandas>=2.2,<3
requests>=2.31          # HTTP-виклики до Groq (OpenAI-сумісний ендпойнт)
numpy>=2.0,<3
scikit-learn>=1.5
joblib>=1.4

# ЧОМУ pandas<3. pandas 3.0 перевів рядкові стовпці на dtype 'str' замість
# 'object'. Код L0-репейру (core/exceptions_taxonomy.py) відбирає колонки
# через select_dtypes(include="object"); у pandas 3 це ще працює через
# shim із попередженням, а в pandas 4 перестане — і фузі-виправлення назв
# показників мовчки зламається. Пін тримає поведінку, на якій зафіксовано
# baseline Фази 0.

# ---- парсинг PDF ----
pdfplumber>=0.11
reportlab>=4.0        # лише для генерації демо-фікстур (make_sample_pdf.py)

# camelot-py — основний парсер за специфікацією (розділ 7), fallback на
# pdfplumber спрацьовує автоматично, якщо його немає. Не в основному
# списку, бо тягне ghostscript як системну залежність:
#     pip install "camelot-py[base]"

# ---- UI ----
streamlit>=1.40

# ---- опційно: повний векторний стек (retrieval/vector_index.py) ----
# Без них векторний індекс працює на char-ngram TF-IDF (scikit-learn).
# Вмикається ЯВНО: VKSK_USE_EMBEDDINGS=1
# УВАГА: тягне torch (~2 ГБ). На цьому документі дає ГІРШИЙ recall@1,
# ніж TF-IDF-fallback (74% проти 100%) — вимірювання й пояснення в README.
#     pip install chromadb sentence-transformers

# ---- LLM-провайдер ----
# Groq (безкоштовний тариф, console.groq.com) — єдиний провайдер реальних
# викликів. Окремого SDK НЕ потрібно: використовується прямий HTTP на
# requests до https://api.groq.com/openai/v1. Налаштування:
#     export GROQ_API_KEY=gsk_...
#     export VKSK_LLM_PROVIDER=groq          # дефолт, можна не задавати
#     export VKSK_LLM_MODEL=openai/gpt-oss-120b    # дефолт
#
# anthropic — ПОЗА БЮДЖЕТОМ, гілка лишена в коді як історична, у
